    def list_sessions_for_user(self, user_id: str) -> list[ChatSessionInfo]:
        """Return ChatSessionInfo records for sessions owned by user_id.

        Sessions are partitioned by _metadata[session_id]["user_id"]
        (D-22 + D-27; RESEARCH.md Open Question 5 RESOLVED). first_message_preview
        is the first HumanMessage content truncated to 80 chars, or None when
        the history is empty.
        """
        results: list[ChatSessionInfo] = []
        for session_id, metadata in self._metadata.items():
            if metadata.get("user_id") != user_id:
                continue
            history = self._histories.get(session_id)
            preview: str | None = None
            if history is not None:
                for msg in history.messages:
                    if isinstance(msg, HumanMessage):
                        content = msg.content if isinstance(msg.content, str) else str(msg.content)
                        preview = content[:80]
                        break
            results.append(
                ChatSessionInfo(
                    session_id=session_id,
                    provider=metadata["provider"],
                    model=metadata["model"],
                    created_at=metadata["created_at"],
                    first_message_preview=preview,
                )
            )
        # Newest first — the sidebar is reverse-chronological.
        results.sort(key=lambda info: info.created_at, reverse=True)
        return results
    ```

    @router.post("/api/providers/refresh", response_model=ProviderRefreshResponse)
    async def refresh_providers(
        request: Request,
        _current_user: Annotated[User, Depends(get_current_active_user)],
        factory: Annotated[LLMProviderFactory, Depends(get_llm_factory)],
    ) -> ProviderRefreshResponse:
        """Re-discover all local providers in parallel; return enriched provider list (D-06).

        Cache TTL gating: per-entry, ttl from Settings.provider_models_cache_ttl_seconds.
        On partial unreachable, returns 200 with the unreachable entries marked
        available=false and error="provider_unreachable" — clients show the marker
        rather than failing the whole settings page.
        """
        cache: dict[str, list[str]] = request.app.state.provider_models_cache
        timestamps: dict[str, float] = request.app.state.provider_models_cache_timestamps
        ttl = settings.provider_models_cache_ttl_seconds
        now = time.time()

        # Determine which providers need a fresh discovery vs cache hit.
        local_provider_names = ["ollama"]
        # Only include lmstudio if the provider class is importable (Plan 06 narrowing).
        try:
            from app.llm.providers.lmstudio import LMStudioProvider  # noqa: F401
            local_provider_names.append("lmstudio")
        except ImportError:
            pass

        stale = [
            name for name in local_provider_names
            if name not in timestamps or (now - timestamps[name]) > ttl
        ]

        if stale:
            fresh = await factory.refresh_local_models()
            for name, models in fresh.items():
                if name not in stale:
                    continue  # Skip non-stale even if refresh returned them — preserves TTL.
                if models is None:
                    cache[name] = []
                    timestamps[name] = now
                else:
                    cache[name] = models
                    timestamps[name] = now

        # Build the response from cache. Mark unreachable when cache is empty AND we
        # just refreshed (i.e., the daemon was actually unreachable, not just unprobed).
        entries: list[ProviderRefreshEntry] = []
        for name in local_provider_names:
            models = cache.get(name, [])
            available = bool(models)
            error = None if available else "provider_unreachable"
            entries.append(ProviderRefreshEntry(
                name=name,
                models=models,
                available=available,
                error=error,
            ))

        return ProviderRefreshResponse(providers=entries)
    ```
